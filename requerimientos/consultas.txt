    // Permitir que Enter avance al siguiente campo del formulario (salvo textarea y botones)
    document.addEventListener('DOMContentLoaded', function () {
        const form = document.querySelector('.form-consulta');
        if (!form) return;
        const sel = 'input:not([type="hidden"]):not([disabled]), select:not([disabled]), textarea:not([disabled]), button:not([disabled])';

        form.addEventListener('keydown', function (e) {
            if (e.key !== 'Enter') return;
            const target = e.target;
            // dejar Enter normal en textarea y botones
            if (target && (target.tagName.toLowerCase() === 'textarea' || target.tagName.toLowerCase() === 'button')) return;
            // evitar submit automático
            e.preventDefault();

            const focusables = Array.from(form.querySelectorAll(sel)).filter(el => {
                // filtrar elementos invisibles
                return el.offsetParent !== null || el === document.activeElement;
            });
            let idx = focusables.indexOf(target);
            if (idx === -1) idx = focusables.indexOf(document.activeElement);
            const next = focusables[idx + 1];
            if (next) {
                next.focus();
                try { if (next.select) next.select(); } catch(err){}
            } else {
                const submitBtn = form.querySelector('button[type="submit"], input[type="submit"]');
                if (submitBtn) submitBtn.focus();
            }
        }, true);

        // Mostrar campo colegio si se selecciona "taller colegio" en origen
        const origenSel = document.getElementById('origen');
        const colegioInput = document.getElementById('nombre_colegio');
        const origenOtroInput = document.getElementById('origen_otro');
        if (origenSel && colegioInput && origenOtroInput) {
            origenSel.addEventListener('change', function() {
                if (this.value === 'taller colegio') {
                    colegioInput.style.display = '';
                    colegioInput.required = true;
                    colegioInput.focus();
                } else {
                    colegioInput.style.display = 'none';
                    colegioInput.required = false;
                    colegioInput.value = '';
                }
                if (this.value === 'otro') {
                    origenOtroInput.style.display = '';
                    origenOtroInput.required = true;
                    origenOtroInput.focus();
                } else {
                    origenOtroInput.style.display = 'none';
                    origenOtroInput.required = false;
                    origenOtroInput.value = '';
                }
            });
        }
    });
    // Forzar sólo dígitos en telf; permitir '+' (único, al inicio) en whatsapp
    const elTelf = document.getElementById('telf');
    if (elTelf) {
        elTelf.addEventListener('input', function(){
            const pos = this.selectionStart || 0;
            this.value = this.value.replace(/\D/g, '');
            try { this.setSelectionRange(pos, pos); } catch(e){}
        });
    }
    const elWhats = document.getElementById('whatsapp');
    if (elWhats) {
        elWhats.addEventListener('input', function(){
            const pos = this.selectionStart || 0;
            // eliminar todo lo que no sea dígito o '+'
            let v = this.value.replace(/[^\d+]/g, '');
            // contar '+'; si hay más de una, mantener sólo una
            const plusCount = (v.match(/\+/g) || []).length;
            v = v.replace(/\+/g, '');
            if (plusCount > 0) v = '+' + v;
            this.value = v;
            try { this.setSelectionRange(pos, pos); } catch(e){}
        });
    }
