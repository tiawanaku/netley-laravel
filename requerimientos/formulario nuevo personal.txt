            document.addEventListener('DOMContentLoaded', function() {
                const hiddenInput = document.getElementById('especialidad_hidden');
                const cbAbogado = document.querySelector('input[name="cargo[]"][value="Abogado"]');
                const cbPsico = document.querySelector('input[name="cargo[]"][value="Psicologia"]');
                const cbMedico = document.querySelector('input[name="cargo[]"][value="Medico"]');
                const cbTrabajoSocial = document.querySelector('input[name="cargo[]"][value="Trabajo Social"]');

                const blockLegal = document.getElementById('especialidad_legal');
                const blockPsico = document.getElementById('especialidad_psico');
                const blockMed = document.getElementById('especialidad_medica');
                const blockTS = document.getElementById('especialidad_ts');

                const allBtns = document.querySelectorAll('.btn-especialidad');

                allBtns.forEach(btn => {
                    btn.addEventListener('click', function() {
                        // Determinar a qué grupo pertenece el botón
                        const parent = btn.closest('.especialidad-block');
                        if (!parent) return;
                        const gid = parent.id;
                        // Validar que el cargo correspondiente esté seleccionado
                        if (gid === 'especialidad_legal' && cbAbogado && !cbAbogado.checked) return;
                        if (gid === 'especialidad_psico' && cbPsico && !cbPsico.checked) return;
                        if (gid === 'especialidad_medica' && cbMedico && !cbMedico.checked) return;
                        if (gid === 'especialidad_ts' && cbTrabajoSocial && !cbTrabajoSocial.checked) return;
                        // Multi-select: alternar selección sin deseleccionar otras
                        btn.classList.toggle('selected');
                        actualizarEspecialidad();
                    });
                });

                function actualizarEspecialidad() {
                    const seleccionadas = Array.from(document.querySelectorAll('.btn-especialidad.selected')).map(b => b.getAttribute('data-value'));
                    hiddenInput.value = seleccionadas.join(',');
                }

                function updateVisibility() {
                    // Legal
                    if (cbAbogado && cbAbogado.checked) {
                        blockLegal.style.display = '';
                    } else {
                        // ocultar y limpiar
                        blockLegal.style.display = 'none';
                        blockLegal.querySelectorAll('.btn-especialidad.selected').forEach(b=>b.classList.remove('selected'));
                    }
                    // Psicología
                    if (cbPsico && cbPsico.checked) {
                        blockPsico.style.display = '';
                    } else {
                        blockPsico.style.display = 'none';
                        blockPsico.querySelectorAll('.btn-especialidad.selected').forEach(b=>b.classList.remove('selected'));
                    }
                    // Médicas
                    if (cbMedico && cbMedico.checked) {
                        blockMed.style.display = '';
                    } else {
                        blockMed.style.display = 'none';
                        blockMed.querySelectorAll('.btn-especialidad.selected').forEach(b=>b.classList.remove('selected'));
                    }
                    // Trabajo Social
                    if (cbTrabajoSocial && cbTrabajoSocial.checked) {
                        blockTS.style.display = '';
                    } else {
                        blockTS.style.display = 'none';
                        blockTS.querySelectorAll('.btn-especialidad.selected').forEach(b=>b.classList.remove('selected'));
                    }
                    actualizarEspecialidad();
                }

                // Hooks para checkboxes
                if (cbAbogado) cbAbogado.addEventListener('change', updateVisibility);
                if (cbPsico) cbPsico.addEventListener('change', updateVisibility);
                if (cbMedico) cbMedico.addEventListener('change', updateVisibility);
                if (cbTrabajoSocial) cbTrabajoSocial.addEventListener('change', updateVisibility);

                // Inicializar visibilidad
                updateVisibility();

                // Enviar: si ninguno de los cargos que tienen especialidades está seleccionado, forzar NO APLICA
                const form = document.querySelector('.form-container form');
                if (form) {
                    form.addEventListener('submit', function(e){
                        const hasSpecialityCargo = (cbAbogado && cbAbogado.checked) || (cbPsico && cbPsico.checked) || (cbMedico && cbMedico.checked) || (cbTrabajoSocial && cbTrabajoSocial.checked);
                        const sel = Array.from(document.querySelectorAll('.btn-especialidad.selected')).map(b => b.getAttribute('data-value')).join(',');
                    if (sel) {
                        hiddenInput.value = sel;
                    } else {
                        if (!hasSpecialityCargo) {
                            hiddenInput.value = 'NO APLICA';
                        } else {
                            hiddenInput.value = '';
                        }
                    }
                    });
                }
             });

<script>
// Permitir que Enter avance al siguiente campo del formulario (salvo en textarea)
document.addEventListener('DOMContentLoaded', function () {
    const form = document.querySelector('.form-container form');
    if (!form) return;
    const sel = 'input:not([type="hidden"]):not([disabled]), select:not([disabled]), textarea:not([disabled]), button:not([disabled])';

    form.addEventListener('keydown', function (e) {
        if (e.key !== 'Enter') return;
        const target = e.target;
        // dejar Enter normal en textarea
        if (target && target.tagName && target.tagName.toLowerCase() === 'textarea') return;
        // evitar submit automático
        e.preventDefault();

        const focusables = Array.from(form.querySelectorAll(sel)).filter(el => {
            // filtrar elementos invisibles
            return el.offsetParent !== null || el === document.activeElement;
        });
        let idx = focusables.indexOf(target);
        if (idx === -1) idx = focusables.indexOf(document.activeElement);
        const next = focusables[idx + 1];
        if (next) {
            // si es un botón de especialidad, "click" y mantener foco
            if (next.classList.contains('btn-especialidad')) {
                next.click();
                next.focus();
            } else {
                next.focus();
                // seleccionar el contenido si es input de texto
                try { if (next.select) next.select(); } catch(err){}
            }
        } else {
            // si no hay siguiente control, intentar enfocar el botón submit si existe
            const submitBtn = form.querySelector('button[type="submit"], input[type="submit"]');
            if (submitBtn) submitBtn.focus();
        }
    }, true);
});
</script>

    // Modal creado bajo demanda + lógica de re-check con cuenta regresiva (10s)
    const repeatState = { ci: { value:'', timeoutId:null, intervalId:null, remaining:0 }, telf: { value:'', timeoutId:null, intervalId:null, remaining:0 } };

    function createModalIfNeeded() {
        let overlay = document.getElementById('dup-modal');
        if (!overlay) {
            overlay = document.createElement('div');
            overlay.id = 'dup-modal';
            overlay.className = 'modal-overlay';
            overlay.innerHTML = `
                <div class="modal-box" role="dialog" aria-modal="true">
                    <div class="modal-header"><h3 id="modal-title"></h3></div>
                    <div class="modal-body" id="modal-body"></div>
                    <div class="modal-actions">
                        <button id="modal-close">Cerrar</button>
                    </div>
                </div>`;
            document.body.appendChild(overlay);
            document.getElementById('modal-close').addEventListener('click', function(){ 
                overlay.style.display = 'none';
                const t = overlay.dataset.type;
                const v = overlay.dataset.value;
                if (t && v) startRecheckTimer(t, v);
            });
            overlay.addEventListener('click', function(e){ if (e.target === overlay) { overlay.style.display = 'none'; const t = overlay.dataset.type; const v = overlay.dataset.value; if (t && v) startRecheckTimer(t, v); } });
        }
        return overlay;
    }

    function showModalImmediate(title, msg, type, value) {
        const overlay = createModalIfNeeded();
        overlay.dataset.type = type || '';
        overlay.dataset.value = value || '';
        document.getElementById('modal-title').textContent = title;
        document.getElementById('modal-body').textContent = msg;
        overlay.style.display = 'flex';
        const btn = document.getElementById('modal-close'); if (btn) btn.focus();
    }

    function clearRepeat(type) {
        const s = repeatState[type];
        if (!s) return;
        if (s.timeoutId) { clearTimeout(s.timeoutId); s.timeoutId = null; }
        if (s.intervalId) { clearInterval(s.intervalId); s.intervalId = null; }
        s.value = '';
        s.remaining = 0;
        if (type === 'ci' && typeof ciWarning !== 'undefined' && ciWarning) { ciWarning.style.display = 'none'; ciWarning.textContent = ''; }
        if (type === 'telf' && typeof telfWarning !== 'undefined' && telfWarning) { telfWarning.style.display = 'none'; telfWarning.textContent = ''; }
    }

    function startRecheckTimer(type, value) {
        clearRepeat(type);
        const s = repeatState[type];
        s.value = value;
        s.remaining = 10;
        const win = type === 'ci' ? ciWarning : telfWarning;
        if (win) { win.style.display = ''; win.textContent = `Tienes ${s.remaining} segundos para cambiarlo...`; }
        s.intervalId = setInterval(function(){
            s.remaining -= 1;
            if (win) win.textContent = `Tienes ${s.remaining} segundos para cambiarlo...`;
            if (s.remaining <= 0) {
                clearRepeat(type);
                const current = (type === 'ci' ? (ciInput ? ciInput.value.trim() : '') : (telfInput ? telfInput.value.replace(/\D/g,'') : ''));
                if (current === value) {
                    (async function(){
                        try {
                            const res = await checkExists(type === 'ci' ? current : '', type === 'telf' ? current : '');
                            if ((type === 'ci' && res.ci_exists) || (type === 'telf' && res.telf_exists)) {
                                showModalImmediate(type === 'ci' ? 'CI duplicado' : 'Teléfono duplicado', type === 'ci' ? 'El carnet de identidad sigue registrado.' : 'El número de teléfono sigue registrado.', type, value);
                            }
                        } catch (e) { console.warn(e); }
                    })();
                }
            }
        }, 1000);
    }

    async function checkExists(ci, telf) {
        const params = new URLSearchParams();
        if (ci) params.set('ci', ci);
        if (telf) params.set('telf', telf.replace(/\D/g,''));
        const res = await fetch('check_personal.php?' + params.toString(), { cache: 'no-store' });
        return res.json();
    }

    // Expedido: mostrar campo cuando se selecciona 'OTRO'
    if (expedidoSelect && expedidoOtro) {
        function updateExpedido() {
            if (expedidoSelect.value === 'OTRO') {
                expedidoOtro.style.display = '';
                expedidoOtro.required = true;
                try { expedidoOtro.focus(); } catch (e) {}
            } else {
                expedidoOtro.style.display = 'none';
                expedidoOtro.required = false;
                expedidoOtro.value = '';
            }
        }
        expedidoSelect.addEventListener('change', updateExpedido);
        updateExpedido();
    }

    if (ciInput) {
        ciInput.addEventListener('input', function() { ciAlertedFor = ''; });
        ciInput.addEventListener('blur', async function() {
            const v = this.value.trim();
            if (!v) return;
            if (ciAlertedFor === v) return; // ya se mostró alerta para este valor
            try {
                const r = await checkExists(v, '');
                if (r.ci_exists) { showModal('CI duplicado', 'El carnet de identidad ya está registrado.'); ciAlertedFor = v; this.focus(); }
            } catch (err) { console.warn(err); }
        });
    }

    if (telfInput) {
        telfInput.addEventListener('input', function() { telfAlertedFor = ''; });
        telfInput.addEventListener('blur', async function() {
            const v = this.value.replace(/\D/g,'');
            if (!v) return;
            if (telfAlertedFor === v) return; // ya se mostró alerta para este valor
            try {
                const r = await checkExists('', v);
                if (r.telf_exists) { showModal('Teléfono duplicado', 'El número de teléfono ya está registrado.'); telfAlertedFor = v; this.focus(); }
            } catch (err) { console.warn(err); }
        });
    }

    if (form) {
        form.addEventListener('submit', async function(e) {
            e.preventDefault();
            const ci = ciInput ? ciInput.value.trim() : '';
            const telf = telfInput ? telfInput.value.replace(/\D/g,'') : '';
            try {
                const r = await checkExists(ci, telf);
                if (r.ci_exists) {
                    if (ciAlertedFor !== ci) { showModal('CI duplicado', 'El carnet de identidad ya está registrado.'); ciAlertedFor = ci; }
                    if (ciInput) ciInput.focus();
                    return;
                }
                if (r.telf_exists) {
                    if (telfAlertedFor !== telf) { showModal('Teléfono duplicado', 'El número de teléfono ya está registrado.'); telfAlertedFor = telf; }
                    if (telfInput) telfInput.focus();
                    return;
                }
                form.submit();
            } catch (err) { console.error(err); form.submit(); }
        });
    }
});
