                        document.getElementById('legal_extra').addEventListener('change', function() {
                            var materia = this.value;
                            var box = document.getElementById('delitos-materia-box');
                            var select = document.getElementById('delito_materia');
                            if (!materia) {
                                box.style.display = 'none';
                                select.innerHTML = '<option value="">Seleccione materia legal primero...</option>';
                                return;
                            }
                            fetch('api_delitos_materia.php?materia=' + encodeURIComponent(materia))
                                .then(r => r.json())
                                .then(data => {
                                    if (data.delitos && data.delitos.length > 0) {
                                        let html = '<option value="">Seleccione delito...</option>';
                                        data.delitos.forEach(function(d) {
                                            html += `<option value=\"${d.id}\">${d.nombre}</option>`;
                                        });
                                        select.innerHTML = html;
                                        box.style.display = '';
                                    } else {
                                        select.innerHTML = '<option value="">No hay delitos para esta materia</option>';
                                        box.style.display = '';
                                    }
                                });
                        });

                            var final = document.getElementById('designacion_final');
                            var delitoInput = document.getElementById('delito');
                            
                            if (designacion.value === 'Legal' && legal.value) {
                                final.value = legal.value;
                                // Guardar delito por separado si existe
                                if (delito && delito.value) {
                                    var delitoText = delito.options[delito.selectedIndex].text;
                                    delitoInput.value = delitoText;
                                } else {
                                    delitoInput.value = '';
                                }
                            } else {
                                final.value = designacion.value;
                                delitoInput.value = '';
                            }
                        });

                        (function(){

                                if (designacion && designacion.value === 'Legal') {
                                    return (legal && legal.value) ? legal.value : designacion.value;

                                return designacion ? designacion.value : '';

                            function getDelitoTexto() {
                                // Preferir el texto del select solo si el option seleccionado tiene un value válido (no placeholder)
                                if (!delitoSelect) return initialDelitoText || '';
                                const idx = delitoSelect.selectedIndex;
                                if (idx >= 0 && delitoSelect.options[idx]) {
                                    const opt = delitoSelect.options[idx];
                                    const val = (opt.value || '').toString().trim();
                                    if (val !== '') {
                                        const txt = opt.text.trim();
                                        if (txt) return txt;
                                    }
                                }
                                // Si no hay selección válida, usar el delito inicial (última respuesta/informe)
                                return initialDelitoText || '';
                            }

                            function getDelitoId() {
                                // Usar valor del select solo si es numérico (id), o usar data-delito-id si está presente
                                if (!delitoSelect) return initialDelitoId || '';
                                const idx = delitoSelect.selectedIndex;
                                if (idx >= 0 && delitoSelect.options[idx]) {
                                    const opt = delitoSelect.options[idx];
                                    const val = (opt.value || '').toString().trim();
                                    if (val !== '' && !isNaN(val)) return val; // numeric likely id
                                    const attr = opt.getAttribute('data-delito-id');
                                    if (attr) return attr;
                                }
                                // Fallback al id detectado en servidor
                                return initialDelitoId || '';
                            }

                            function buildUrl() {
                                // Si hay cliente existente, llevar directamente a su ficha
                                if (existingClienteId) return 'nuevo_caso.php?id_cliente=' + encodeURIComponent(existingClienteId) + '&id_con=28';

                                const base = 'ascender_caso.php?id_con=28';
                                const especialidad = getEspecialidadFinal() || '';
                                const delito = getDelitoTexto() || '';
                                const delitoId = getDelitoId() || '';
                                const ciudad = ciudadForAsc || '';
                                let url = base;
                                // Incluir parámetros siguiendo el formato esperado por ascender_caso.php
                                if (especialidad) url += '&materia_legal=' + encodeURIComponent(especialidad);
                                if (delito) url += '&delito=' + encodeURIComponent(delito);
                                if (delitoId) url += '&delito_select=' + encodeURIComponent(delitoId);
                                if (ciudad) url += '&ciudad=' + encodeURIComponent(ciudad);
                                if (nombresForAsc) url += '&nombres=' + encodeURIComponent(nombresForAsc);
                                if (apellidosForAsc) url += '&apellidos=' + encodeURIComponent(apellidosForAsc);
                                if (telefonoForAsc) url += '&telefono=' + encodeURIComponent(telefonoForAsc);
                                if (latestInformeId) url += '&id_informe=' + encodeURIComponent(latestInformeId);
                                // Incluir prefill del personal encargado si lo tenemos desde el servidor
                                if (prefPersonalIdFromServer) url += '&pref_personal_id=' + encodeURIComponent(prefPersonalIdFromServer);
                                else if (prefPersonalNameFromServer) url += '&pref_personal_name=' + encodeURIComponent(prefPersonalNameFromServer);
                                return url;
                            }

                            // Actualiza el href dinámicamente (opcional), y en click forzamos navegación usando los valores actuales
                            function updateHref() {
                                try { btnAsc.href = buildUrl(); } catch(e){ console.error(e); }
                            }

                            document.addEventListener('change', updateHref);
                            // Also update on load
                            updateHref();

                            btnAsc.addEventListener('click', function(e){
                                e.preventDefault();
                                const url = buildUrl();
                                // Si la especialidad no está definida, dejamos pasar solo con id_con (o si es ficha: también está bien)
                                if (url) window.location.href = url;
                            });
                        })();
